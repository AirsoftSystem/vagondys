// Exemple d'utilisation dans OfficialButton.tsx
import { useAudioControl, COUNTDOWN_SEQUENCE } from "@/lib/hooks/useAudioControl";

export function OfficialButton({ action, onAction }: OfficialButtonProps) {
  const { isPlaying, playCountdown } = useAudioControl({
    onStart: () => console.log("Décompte démarré"),
    onComplete: () => console.log("Décompte terminé, envoi de la commande"),
    onError: (err) => console.error("Erreur audio:", err),
  });

  const handleClick = async () => {
    // 1. Jouer le décompte vocal
    await playCountdown(COUNTDOWN_SEQUENCE);
    
    // 2. Envoyer la commande après le décompte
    onAction();
  };

  return (
    <button
      onClick={handleClick}
      disabled={isPlaying}
      className={isPlaying ? "opacity-50" : ""}
    >
      {isPlaying ? "DÉCOMPTE..." : action === "START_GAME" ? "DÉMARRER" : "VALIDER"}
    </button>
  );
}
